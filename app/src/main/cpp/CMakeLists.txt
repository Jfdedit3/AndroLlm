cmake_minimum_required(VERSION 3.22)
project(androllm LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../../../llama.cpp ${CMAKE_CURRENT_BINARY_DIR}/llama)
add_library(androllm SHARED native.cpp)
target_include_directories(androllm PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../../../llama.cpp/include)
target_link_libraries(androllm PRIVATE llama log)
